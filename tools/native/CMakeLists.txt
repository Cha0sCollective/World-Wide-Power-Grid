cmake_minimum_required(VERSION 3.16)
project(WWPGGuardedNative LANGUAGES C)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)

# Use the exact PG/SuperLU/BLAS revisions in sources.lock.json. The two small
# patches change error handling and ownership, not the electrical equations.
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 11)
set(BUILD_SHARED_LIBS OFF)
set(BUILD_STATIC_LIBS ON)
set(BUILD_WITHOUT_LAPACK ON)
set(BUILD_TESTING OFF)
set(BUILD_SINGLE ON)
set(BUILD_DOUBLE ON)
set(BUILD_COMPLEX OFF)
set(BUILD_COMPLEX16 OFF)
set(BUILD_MATGEN OFF)
set(USE_THREAD OFF)
set(NOFORTRAN ON)
set(TARGET NEHALEM)
set(DYNAMIC_ARCH ON)
set(DYNAMIC_LIST "BULLDOZER;ZEN;SKYLAKEX")
add_subdirectory("${OPENBLAS_SOURCE}" openblas)
export(TARGETS openblas_static FILE "staticblas.cmake")

set(TPL_BLAS_LIBRARIES openblas_static CACHE STRING "Pinned BLAS target" FORCE)
set(enable_internal_blaslib OFF)
set(enable_single OFF)
set(enable_double ON)
set(enable_complex OFF)
set(enable_complex16 OFF)
set(enable_matlabmex OFF)
set(enable_doc OFF)
set(enable_examples OFF)
set(enable_fortran OFF)
set(enable_tests OFF)
add_subdirectory("${SUPERLU_SOURCE}" superlu)
target_compile_options(superlu PRIVATE -mavx2)

set(PG_C "${PG_SOURCE}/native/src/powergridNative/c")
add_library(powergridNative SHARED "${PG_C}/sparse.c" "${PG_C}/solver.c"
    "${PG_C}/mna.c" "${PG_C}/exception_handler.c" "${PG_C}/verify.c")
set_target_properties(powergridNative PROPERTIES OUTPUT_NAME powergridNative7)
target_compile_options(powergridNative PRIVATE -mavx2)
set_source_files_properties("${PG_C}/exception_handler.c" "${PG_C}/verify.c"
    PROPERTIES COMPILE_OPTIONS -mno-avx2)
target_include_directories(powergridNative PRIVATE
    "${PG_SOURCE}/native/src/powergridNative/include" "${JNI_INCLUDE}" "${JNI_PLATFORM_INCLUDE}")
target_link_libraries(powergridNative PRIVATE superlu openblas_static)
if(WIN32)
    target_link_options(powergridNative PRIVATE -static-libgcc -static)
    set_target_properties(powergridNative PROPERTIES PREFIX "lib")
endif()

add_executable(sparse_safety_tests "${CMAKE_CURRENT_SOURCE_DIR}/sparse_safety_tests.c" "${PG_C}/sparse.c")
target_include_directories(sparse_safety_tests PRIVATE "${PG_SOURCE}/native/src/powergridNative/include")
target_link_libraries(sparse_safety_tests PRIVATE superlu openblas_static)
target_link_options(sparse_safety_tests PRIVATE -Wl,--wrap=dgstrf -Wl,--wrap=dgstrs -Wl,--wrap=dsnode_dfs)
target_compile_options(sparse_safety_tests PRIVATE -UNDEBUG)
if(WIN32)
    target_link_options(sparse_safety_tests PRIVATE -static-libgcc -static)
endif()
